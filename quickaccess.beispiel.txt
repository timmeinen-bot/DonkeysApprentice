# Donkey's Apprentice -- example list
#
# This file ships with the package. On the FIRST start it is turned into
# quickaccess.txt; from then on only quickaccess.txt is read and this
# example file is never touched again.
#
# Format:
#   [Group]                    becomes a heading
#   Name = Target              file, folder or link
#   Name = Target | Source,Nr  custom icon, e.g. shell32.dll,44
#   Target                     without "=": the file name becomes the name
#
# Environment variables in percent signs are expanded, so the list works on
# any machine. Targets starting with "shell:" are opened by Explorer
# (special locations such as the recycle bin).

[Start]
Documents = %USERPROFILE%\Documents
Downloads = %USERPROFILE%\Downloads
Pictures = %USERPROFILE%\Pictures
Recycle Bin = shell:RecycleBinFolder

[Windows]
Settings = shell:::{ED7BA470-8E54-465E-825C-99712043E01C}
Programs and Features = shell:ChangeRemoveProgramsFolder
Startup = shell:Startup

[On the web]
Search = https://duckduckgo.com
Weather = https://www.dwd.de
Timetable = https://www.bahn.de
